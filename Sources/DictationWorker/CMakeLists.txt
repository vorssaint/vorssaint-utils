cmake_minimum_required(VERSION 3.20)
project(VorssaintDictation LANGUAGES C CXX OBJC OBJCXX)
if(NOT TRANSCRIBE_SOURCE)
  message(FATAL_ERROR "Set TRANSCRIBE_SOURCE to the pinned transcribe.cpp checkout")
endif()
set(CMAKE_CXX_STANDARD 17)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
include_directories("${TRANSCRIBE_SOURCE}/include")
add_subdirectory("${TRANSCRIBE_SOURCE}" transcribe)
add_executable(VorssaintDictationWorker main.mm)
target_compile_options(VorssaintDictationWorker PRIVATE -fobjc-arc)
target_link_libraries(VorssaintDictationWorker PRIVATE transcribe "-framework Foundation")
